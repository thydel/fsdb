# Make a better smaller DB

- Aggregate by hours
- Use `timestamp` instead `unit64` for UTS
- Keep only appli, node and aggregated stat over size by hour

```yaml
- id: hourly-db
  jq: |
    "attach '\($out)' as out;
    create table out.hours as select date_trunc('hour', to_timestamp(uts)) as start, appli, node,
    sum(size) sum, count(*) cnt, min(size) min, max(size) max, avg(size) avg
    from stat where uts > 0 and size != 0 group by 1, 2, 3 order by start"
  sh: |
    self -nr --arg out ${1:-out/tdep-hours.db}
  tt: |-
    hourly-db | duckdb tmp/tdep.db
    duckdb tmp/tdep-hours.db --csv -c 'select * from hours' > out/tdep-hours.csv
```

## Define `table-last` pipeline

- And show an example of SQL generated by bigger pipelines

```bash
pivot () { : ${3:?}; start | span-last $1 $2 | resample $2 | show $3 | add-row-total appli | pivot-span appli total; }
filter () { move-last | limit-cumul --pct=.99 --col=total | add-sum-row | to-unit $4; }
table-last () { pivot "$@" | filter "$@"; }
```

```bash
table-last 6 month %y-%m | merge-cte
```

```sql
WITH step0 AS (FROM hours),
step1 AS (select * from step0 where start >= (select max(start) from step0) - interval 6 month),
step2 AS (select 
   date_trunc('month', start) as start, appli, node,
   sum(sum) as sum, sum(cnt) as cnt,  min(min) as min,
   max(max) as max, sum(sum) / sum(cnt) as avg
from step1 group by 1, 2, 3),
step3 AS (select strftime(start, '%y-%m') as span, * from step2),
step4 AS (SELECT *, sum(sum) OVER (partition by appli) as total FROM step3),
step5 AS (pivot step4 on span using sum(sum) group by appli, total),
step6 AS (select * exclude(total), total from step5),
step7 AS (select * from step6 
qualify (sum(total) over (order by total desc) - total) / sum(total) over () < .99),
step8 AS (select * from step7 union all
select coalesce(cast(sum(try_cast(columns(*) as hugeint)) as varchar), ' TOTAL')
from step7),
step9 AS (select
  case
    when try_cast(columns(*) as double) is not null
    then cast(round(try_cast(columns(*) as double) / pow(2.0, 30), 2) as varchar)
    else columns(*)
  end from step8)
SELECT * FROM step9
```
